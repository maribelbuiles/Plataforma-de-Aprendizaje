import streamlit as st

st.set_page_config(page_title="Mi Plataforma de Aprendizaje", page_icon="🎓")

st.write("Esta es una aplicación base para gestionar cursos y contenido educativo.")

    st.header("Bienvenido al curso")
    st.write("Aquí puedes empezar tu viaje de aprendizaje.")

    st.info("Contenido del módulo 1 cargado correctamente.")
